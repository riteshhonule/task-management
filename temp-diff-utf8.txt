+                      {selectedTaskDetail.status?.replace('_', ' ')}
+                    </span>
+                  </div>
+
+                  <div className="bg-slate-50 border border-slate-100 rounded-xl p-3">
+                    <span className="block text-[10px] font-bold text-slate-450 uppercase mb-0.5">Completion Percentage</span>

+                  </div>
+                </div>
+
+                {selectedTaskDetail.completedWorkDescription && (
+                  <div className="bg-emerald-50/50 border border-emerald-100 rounded-xl p-4 text-xs text-emerald-900 text-left">
+                    <strong className="block text-emerald-800 mb-1">Completed Work:</strong>
+                    <p className="leading-relaxed whitespace-pre-wrap">{selectedTaskDetail.completedWorkDescription}</p>

+
+                {selectedTaskDetail.delayReason && (
+                  <div className="bg-rose-50/50 border border-rose-100 rounded-xl p-4 text-xs text-rose-900 text-left">
+                    <strong className="block text-rose-850 mb-1">Delay Reason:</strong>
+                    <p className="leading-relaxed whitespace-pre-wrap">{selectedTaskDetail.delayReason}</p>
+                  </div>

+
+                {selectedTaskDetail.blockedReason && (
+                  <div className="bg-rose-50/50 border border-rose-100 rounded-xl p-4 text-xs text-rose-900 text-left">
+                    <strong className="block text-rose-850 mb-1">Blocked Reason:</strong>
+                    <p className="leading-relaxed whitespace-pre-wrap">{selectedTaskDetail.blockedReason}</p>
+                  </div>

+              </div>
+
+              {/* Vertical Chronological Workflow Timeline */}
+              <div className="space-y-4 pt-4 border-t border-slate-100">
+                <h4 className="text-xs font-bold text-slate-400 uppercase tracking-wider font-extrabold">Workflow Timeline</h4>
+                <div className="relative pl-6 border-l border-slate-200 space-y-6">
+                  {selectedTaskDetail.histories?.map((h: any, idx: number) => {
+                    const submission = h.taskSubmissionId
+                      ? selectedTaskDetail.submissions?.find((s: any) => s.id === h.taskSubmissionId)
+                      : null;
+                    
+                    return (
+                      <div key={h.id || idx} className="relative text-left font-sans">
+                        {/* Timeline Dot */}
+                        <div className={`absolute -left-[31px] top-1.5 w-4 h-4 rounded-full border-2 bg-white flex items-center justify-center ${

+                          h.action === 'REVISION_REQUESTED' ? 'border-rose-500 text-rose-500' :
+                          h.action === 'SUBMITTED' ? 'border-indigo-500 text-indigo-500' :
+                          'border-slate-450 text-slate-450'
+                        }`}>
+                          <div className={`w-1.5 h-1.5 rounded-full ${
+                            h.action === 'APPROVED' ? 'bg-emerald-500' :
+                            h.action === 'REVISION_REQUESTED' ? 'bg-rose-500' :
+                            h.action === 'SUBMITTED' ? 'bg-indigo-500' :
+                            'bg-slate-450'
+                          }`}></div>
+                        </div>
+
+                        {/* Timeline Item Content */}
+                        <div className="space-y-1">
+                          <div className="flex justify-between items-center text-xs">
+                            <span className="font-extrabold text-slate-700 uppercase tracking-wider">{h.action.replace('_', ' ')}</span>
+                            <span className="text-slate-400 text-[10px]">{new Date(h.createdAt).toLocaleString()}</span>
+                          </div>
+                          <p className="text-xs text-slate-500 font-semibold">
+                            Performed by: <span className="text-slate-600">{h.performedBy?.name || 'System'}</span>
+                          </p>
+                          {h.details && <p className="text-xs text-slate-600 bg-slate-50 p-2 rounded border border-slate-150">{h.details}</p>}
+                          
+                          {/* Submission Details */}
+                          {submission && (
+                            <div className="mt-2 p-3 bg-slate-50 rounded-xl border border-slate-200 text-xs text-slate-650 space-y-2">
+                              {submission.comment && (
+                                <div>
+                                  <strong>Employee Comment:</strong> <span className="italic">"{submission.comment}"</span>
+                                </div>

+                              {submission.proofs && submission.proofs.length > 0 && (
+                                <div className="space-y-1">
+                                  <strong>Proofs:</strong>
+                                  <div className="flex flex-wrap gap-2 mt-1">
+                                    {submission.proofs.map((proof: any) => (
+                                      <a
+                                        key={proof.id}
+                                        href={uploadsApi.getFileUrl(proof.filepath)}
+                                        target="_blank"
+                                        rel="noreferrer"
+                                        className="inline-flex items-center gap-1 bg-indigo-50 hover:bg-indigo-100 border border-indigo-150 text-indigo-700 px-2 py-1 rounded text-[10px] font-semibold"
+                                      >
+                                        View Proof ({proof.filename})
+                                      </a>
+                                    ))}
+                                  </div>
+                                </div>

+                              {submission.reviewComments && submission.reviewComments.length > 0 && (
+                                <div className="pt-2 border-t border-slate-200 mt-2 space-y-1.5">
+                                  <strong>Reviewer Feedback:</strong>
+                                  {submission.reviewComments.map((rc: any) => (
+                                    <div key={rc.id} className="bg-white p-2 rounded border border-slate-150 text-[11px]">
+                                      <div className="flex justify-between text-slate-405 text-[10px] mb-0.5">

+                          )}
+                        </div>
+                      </div>

+                  {(!selectedTaskDetail.histories || selectedTaskDetail.histories.length === 0) && (

+                </div>
+              </div>
+
+              {/* Submission Controls (for Assigned Employee) */}
+              {(selectedTaskDetail.assignedToUserId === user?.id || selectedTaskDetail.employeeId === user?.id) &&
+               ['PENDING', 'IN_PROGRESS', 'REVISION_REQUIRED'].includes(selectedTaskDetail.status) && (
+                <form onSubmit={handleWorkSubmission} className="space-y-4 pt-6 border-t-2 border-slate-100 bg-indigo-50/20 p-4 rounded-xl border border-indigo-100">

+                      ΓÜá A file attachment is mandatory for this submission.
+                    </div>

+                  <div>

+                      onChange={(e) => setProofFile(e.target.files ? e.target.files[0] : null)}
+                      className="w-full text-xs text-slate-600 file:mr-4 file:py-2 file:px-4 file:rounded-full file:border-0 file:text-xs file:font-semibold file:bg-indigo-100 file:text-indigo-700 hover:file:bg-indigo-200 cursor-pointer"
+                    />
+                  </div>
+                  <div>
+                    <label className="block text-[10px] font-bold text-slate-500 uppercase mb-1.5 tracking-wider">Submission Notes / Comments</label>
+                    <textarea
+                      value={submissionComment}
+                      onChange={(e) => setSubmissionComment(e.target.value)}
+                      placeholder="Detail the work completed, links, or comments for the reviewer..."
+                      className="w-full rounded-xl border border-slate-200 px-3 py-2 text-sm focus:ring-2 focus:ring-indigo-500/20 min-h-[60px]"
+                    />
+                  </div>
+                  <button
+                    type="submit"
+                    disabled={isSubmittingWork}
+                    className="w-full py-2.5 bg-indigo-600 hover:bg-indigo-500 text-white text-xs font-bold rounded-xl flex items-center justify-center gap-2 cursor-pointer shadow-md disabled:opacity-50"
+                  >
+                    {isSubmittingWork ? <Loader2 size={14} className="animate-spin" /> : <CheckCircle2 size={14} />}
+                    Submit Work for Approval
+                  </button>
+                </form>
+              )}
+
+              {/* Reviewer Controls (for Delegator/Admins) */}
+              {(user?.role === 'ADMIN' || user?.role === 'SUPER_ADMIN' || (selectedTaskDetail.assignedByUserId === user?.id && selectedTaskDetail.assignedByUserId !== selectedTaskDetail.assignedToUserId)) &&
+               ['PENDING_REVIEW', 'SUBMITTED'].includes(selectedTaskDetail.status) && (
+                <div className="space-y-4 pt-6 border-t-2 border-slate-100 bg-emerald-50/20 p-4 rounded-xl border border-emerald-100">
+                  <h4 className="text-xs font-bold text-emerald-900 uppercase tracking-wider flex items-center gap-1.5 font-heading">
+                    <CheckSquare size={14} /> Review Submission
+                  </h4>
+                  <div>
+                    <label className="block text-[10px] font-bold text-slate-500 uppercase mb-1.5 tracking-wider">Reviewer Feedback Comment *</label>
+                    <textarea
+                      value={reviewComment}
+                      onChange={(e) => setReviewComment(e.target.value)}
+                      placeholder="Provide revision instructions or approval notes..."
+                      className="w-full rounded-xl border border-slate-200 px-3 py-2 text-sm focus:ring-2 focus:ring-indigo-500/20 min-h-[60px]"
+                    />
+                  </div>
+                  <div className="grid grid-cols-2 gap-3">
+                    <button
+                      type="button"
+                      disabled={isSubmittingReview}
+                      onClick={() => handleReviewAction('REVISION_REQUIRED')}
+                      className="py-2.5 bg-rose-600 hover:bg-rose-700 text-white text-xs font-bold rounded-xl flex items-center justify-center gap-1.5 cursor-pointer shadow-sm disabled:opacity-50 animate-in fade-in"
+                    >
+                      Request Revision
+                    </button>
+                    <button
+                      type="button"
+                      disabled={isSubmittingReview}
+                      onClick={() => handleReviewAction('APPROVED')}
+                      className="py-2.5 bg-emerald-600 hover:bg-emerald-700 text-white text-xs font-bold rounded-xl flex items-center justify-center gap-1.5 cursor-pointer shadow-sm disabled:opacity-50 animate-in fade-in"
+                    >
+                      Approve & Complete Task

+
+            </div>

+              <button

+                className="px-5 py-2 rounded-xl text-xs font-bold bg-white border border-slate-200 text-slate-600 hover:bg-slate-100 transition-colors cursor-pointer"
+              >
+                Close
+              </button>
+            </div>
+          </div>
         </div>
       )}
